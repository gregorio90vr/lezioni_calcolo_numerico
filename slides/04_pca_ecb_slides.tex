\section{Motivazioni e obiettivi (\textasciitilde20 min)}
% ============================================================

\begin{frame}{La curva dei tassi di interesse}
\begin{itemize}
  \item La \textbf{curva dei tassi} associa a ogni scadenza $T$ un tasso $r(T)$: cambia ogni
        giorno per decisioni di politica monetaria, aspettative di inflazione, fattori
        macroeconomici.
  \item Dati: curva \textbf{governativa AAA} dell'area euro, ECB Data Portal, scadenze da
        \textbf{1 a 20 anni}.
  \item \textbf{Spot rate} $r(T)$: tasso implicito nel prezzo di uno zero-coupon bond,
        $P(0,T) = e^{-r(T)\,T}$.
\end{itemize}
\begin{center}
\includegraphics[width=0.62\linewidth,height=0.55\textheight,keepaspectratio]{step0_esempio_curva_tassi.pdf}
\end{center}
\end{frame}

\begin{frame}{Il problema della dimensionalità}
\begin{itemize}
  \item Ogni data di valutazione produce un vettore
        $\mathbf{r} = (r(T_1),\ldots,r(T_n)) \in \R^n$, $n=20$.
  \item In ambito attuariale $n$ può salire a 40--50 scadenze.
  \item Seguire e modellare congiuntamente tutte le componenti è complesso e poco pratico.
  \item[$\Rightarrow$] Serve una rappresentazione \textbf{compatta}: pochi
        \emph{movimenti fondamentali} che spiegano la maggior parte delle variazioni.
\end{itemize}
\end{frame}

\begin{frame}{L'obiettivo: PCA via SVD}
\begin{itemize}
  \item Strumento: \textbf{Principal Component Analysis (PCA)}, implementata tramite la
        \textbf{Singular Value Decomposition (SVD)}.
  \item Permette di \textbf{identificare e quantificare} i movimenti principali della curva.
  \item Risultato anticipato: con soli \textbf{3} fattori si descrive quasi per intero
        l'evoluzione storica della curva --- lo vedremo nel Blocco 2.
\end{itemize}
\end{frame}

\begin{frame}{Struttura della dispensa: calibrazione e proiezione}
\`E prassi consolidata articolare il modello in due fasi:
\begin{itemize}
  \item \textbf{Calibrazione}: i parametri (i loadings PCA) sono stimati sulle variazioni
        storiche fino a una data di riferimento --- tipicamente fine anno solare.
  \item \textbf{Proiezione}: i parametri vengono \emph{congelati} e usati su date successive,
        fuori dal campione di stima.
\end{itemize}
\begin{block}{In questa lezione}
Calibrazione: \textbf{ottobre 2004 -- 31 dicembre 2025}. Proiezione su due curve fuori
campione: \textbf{31 marzo 2026} (+3 mesi) e \textbf{30 giugno 2026} (+6 mesi).
\end{block}
\end{frame}

\section{La Decomposizione ai Valori Singolari, SVD (\textasciitilde25 min)}
% ============================================================

\begin{frame}{Teorema: esistenza e unicità della SVD}
\begin{teorema}[SVD]
Per ogni matrice $A \in \R^{m\times n}$ esistono:
\begin{itemize}
  \item una matrice ortogonale $U \in \R^{m\times m}$ (\textbf{vettori singolari sinistri});
  \item una matrice ortogonale $V \in \R^{n\times n}$ (\textbf{vettori singolari destri});
  \item una matrice diagonale $\Sigma \in \R^{m\times n}$ con
        $\sigma_1 \ge \sigma_2 \ge \cdots \ge \sigma_{\min(m,n)} \ge 0$
        (\textbf{valori singolari});
\end{itemize}
tali che $A = U\Sigma V^T$. I valori singolari sono unici; i vettori singolari sono unici a
meno di segno quando i $\sigma_i$ sono distinti.
\end{teorema}
\end{frame}

\begin{frame}{Interpretazione geometrica}
Esempio: $A = \begin{pmatrix}1&1\\0&1\end{pmatrix}$ (shear). La SVD scompone l'azione di $A$
in tre trasformazioni:
\begin{itemize}
  \item $V^T$ \textbf{ruota} il cerchio: $\mathbf{v}_1,\mathbf{v}_2$ diventano gli assi
        canonici;
  \item $\Sigma$ \textbf{deforma} lungo gli assi per i fattori $\sigma_1\approx 1{,}618$,
        $\sigma_2\approx 0{,}618$;
  \item $U$ \textbf{ruota} l'ellisse risultante nella posizione finale.
\end{itemize}
\begin{center}
\includegraphics[width=0.75\linewidth,height=0.5\textheight,keepaspectratio]{svd_geometria_esempio.pdf}
\end{center}
\end{frame}

\texttt{svd(A)}, ispezionare \texttt{\$u}, \texttt{\$d}, \texttt{\$v}.}

\section{Norma di Frobenius e Teorema di Eckart--Young--Mirsky (\textasciitilde20 min)}
% ============================================================

\begin{frame}{La norma di Frobenius}
\[
  \|A\|_F = \sqrt{\sum_{i,j} a_{ij}^2} = \sqrt{\operatorname{tr}(A^T A)}.
\]
\begin{proposizione}[Frobenius e valori singolari]
Per ogni matrice $A$ con valori singolari $\sigma_1,\ldots,\sigma_r$ ($r=\operatorname{rk}(A)$):
\[
  \|A\|_F^2 = \sigma_1^2 + \sigma_2^2 + \cdots + \sigma_r^2.
\]
\end{proposizione}
\end{frame}

\begin{frame}{Teorema di Eckart--Young--Mirsky (1936)}
Sia $A = U\Sigma V^T$ e $A_k = \sum_{i=1}^k \sigma_i\,\mathbf{u}_i\mathbf{v}_i^T$ la
\textbf{SVD troncata} alle prime $k$ componenti.
\begin{teorema}[Eckart--Young--Mirsky]
$A_k$ è la \textbf{miglior approssimazione di rango $k$} in norma di Frobenius:
\[
  A_k = \operatorname*{argmin}_{\operatorname{rk}(B)\le k} \|A-B\|_F,
  \qquad
  \|A-A_k\|_F^2 = \sigma_{k+1}^2+\cdots+\sigma_r^2.
\]
\end{teorema}
\begin{alertblock}{Il filo conduttore}
``Miglior approssimazione'' (Eckart--Young) $\;\equiv\;$ ``massima varianza spiegata''
(Frobenius $=$ varianza) $\Rightarrow$ è la base della PCA.
\end{alertblock}
\end{frame}

\section{La PCA come SVD della matrice centrata (\textasciitilde25 min)}
% ============================================================

\begin{frame}{Impostazione}
Matrice $X \in \R^{m\times n}$:
\begin{itemize}
  \item $m$ righe = osservazioni (mesi);
  \item $n$ colonne = variabili osservate (scadenze/maturity).
\end{itemize}
\begin{definizione}[Matrice centrata]
\[
  \tilde X = X - \mathbf{1}_m\,\bar{\mathbf{x}}^T,
  \qquad \bar{\mathbf{x}} \in \R^n \text{ vettore delle medie di colonna}.
\]
\end{definizione}
Centrare isola le \textbf{direzioni di variabilità} dalla posizione media del campione.
\end{frame}

\begin{frame}{Varianza campionaria e norma di Frobenius}
\[
  \mathrm{Var}_{\mathrm{tot}} = \frac{1}{m-1}\sum_{i,j}\tilde x_{ij}^2
  = \frac{\|\tilde X\|_F^2}{m-1}.
\]
Quindi $\|\tilde X\|_F^2 = (m-1)\cdot \mathrm{Var}_{\mathrm{tot}}$: la norma di Frobenius
rappresenta la \textbf{varianza totale}, a meno di una costante.
\end{frame}

\begin{frame}{Teorema: Principal Component Analysis}
Sia $\tilde X = U\Sigma V^T$ la SVD della matrice centrata.
\begin{teorema}[PCA]
\begin{enumerate}
  \item I \textbf{loadings} (direzioni principali) sono le colonne di $V$:
        $\mathbf{v}_1,\mathbf{v}_2,\ldots$
  \item Gli \textbf{scores} (coordinate nelle nuove direzioni) sono le righe di $U\Sigma$,
        cioè $\tilde X V$.
  \item La varianza spiegata dalla componente $i$ è $\sigma_i^2/\sum_j\sigma_j^2$.
\end{enumerate}
\end{teorema}
\end{frame}

\begin{frame}{Quota di varianza spiegata}
Per il Teorema di Eckart--Young, $\tilde X_k$ (SVD troncata) è la miglior approssimazione di
rango $k$, con errore residuo $\|\tilde X - \tilde X_k\|_F^2 = \sigma_{k+1}^2+\cdots+\sigma_n^2$.
\[
  \eta_k = \frac{\|\tilde X_k\|_F^2}{\|\tilde X\|_F^2}
  = \frac{\sigma_1^2+\cdots+\sigma_k^2}{\sigma_1^2+\cdots+\sigma_n^2}
  = 1 - \frac{\sigma_{k+1}^2+\cdots+\sigma_n^2}{\sigma_1^2+\cdots+\sigma_n^2} \;\in [0,1].
\]
\end{frame}

\begin{frame}{Ricostruzione a $k$ componenti}
La $s$-esima riga di $\tilde X_k = U_k\Sigma_k V_k^T$:
\[
  (\tilde X_k)_{s,:} = \hat{\tilde{\mathbf{x}}}_s^{(k)}
  = \sum_{i=1}^{k} \alpha_{s,i}\,\mathbf{v}_i \;\in\; \R^n,
  \qquad \alpha_{s,i} = \sigma_i\,u_{s,i} = (U\Sigma)_{s,i}.
\]
\begin{itemize}
  \item I loadings $\mathbf{v}_i$ indicano le \textbf{direzioni};
  \item gli scores $\alpha_{s,i}$ indicano l'\textbf{ampiezza} dello spostamento.
\end{itemize}
Duplice uso: approssimare una variazione storica con $k$ dimensioni, oppure proiettare una
curva \emph{qualsiasi} su uno spazio di dimensione $k$ (Blocco 2, calibrazione).
\end{frame}

\begin{frame}{Riepilogo Blocco 1}
\begin{enumerate}
  \item \textbf{SVD}: fattorizza qualsiasi matrice in componenti ordinate per importanza.
  \item \textbf{Eckart--Young}: la SVD troncata è la miglior approssimazione di rango $k$
        in norma di Frobenius.
  \item \textbf{Frobenius $=$ varianza}: la stessa proprietà si legge in termini statistici.
  \item \textbf{PCA} $=$ SVD della matrice \emph{centrata}: loadings $=$ colonne di $V$,
        scores $=$ righe di $U\Sigma$.
\end{enumerate}
\vspace{6pt}
\centering\Large Domande prima della pausa?
\end{frame}

\section{Dati ECB e aggregazione mensile (\textasciitilde15 min)}
% ============================================================

\begin{frame}{Perché variazioni mensili?}
\begin{itemize}
  \item Dati ECB disponibili: \textbf{giornalieri}. Ma l'obiettivo non è catturare movimenti
        di brevissimo periodo, bensì l'evoluzione strutturale della curva.
  \item Le variazioni \textbf{giornaliere} sono più adatte a trading/gestione ad alta
        frequenza; le \textbf{mensili} colgono meglio gli eventi macro-finanziari.
  \item Prassi consolidata (Litterman--Scheinkman 1991, Meucci 2009): la PCA si applica alle
        \textbf{variazioni}, non ai livelli.
\end{itemize}
\[
  R^{\mathrm{mens}}_{s,j} = R_{t^\ast(s),j}, \qquad t^\ast(s) = \max\{t : t \in \text{mese } s\}.
\]
Campione di calibrazione: $R^{\mathrm{mens}} \in \R^{256\times 20}$ (set 2004 -- dic 2025).
\end{frame}

\begin{frame}{Gli scores: quanto si muove ciascun fattore}
\[
  \alpha_{s,i} = \sigma_i\,u_{s,i}, \qquad i=1,2,3.

\begin{itemize}
  \item $\alpha_{s,i}$ misura \emph{di quanto} la curva si è mossa nel mese $s$ nella

  \item Valore grande in modulo $\to$ quel fattore ha dominato il movimento; vicino a zero
        $\to$ fattore inattivo.
  \item Gli scores, letti nel tempo, raccontano la storia della politica monetaria BCE.
\end{itemize}
\end{frame}

\section{Calibrazione su curve fuori campione (\textasciitilde20 min)}
% ============================================================

\begin{frame}{L'idea: proiezione ortogonale}
I loadings $\mathbf{v}_1,\ldots,\mathbf{v}_n$ formano una base ortonormale di $\R^n$
($V^TV=I_n$). Data una curva \emph{qualsiasi} $\mathbf{y}\in\R^n$:
\begin{block}{Passo 1 --- Scores}
\[
  \boldsymbol{\alpha}_k = V_k^T\mathbf{y} = (\mathbf{v}_1^T\mathbf{y},\ldots,\mathbf{v}_k^T\mathbf{y})^T.
\]
\end{block}
\begin{block}{Passo 2 --- Ricostruzione}
\[
  \hat{\mathbf{y}}^{(k)} = V_k\boldsymbol{\alpha}_k = \alpha_1\mathbf{v}_1+\cdots+\alpha_k\mathbf{v}_k.
\]
\end{block}
\end{frame}

\begin{frame}{Forma compatta: il proiettore ortogonale}
\[
  \hat{\mathbf{y}}^{(k)} = V_k(V_k^T\mathbf{y}) = (V_kV_k^T)\,\mathbf{y}, \qquad
  P_k = V_kV_k^T \in \R^{n\times n}.
\]
\begin{itemize}
  \item $P_k$ è il proiettore ortogonale sullo spazio generato dalle prime $k$ colonne di $V$.
  \item Per $k=n$: $V_nV_n^T=I_n$, ricostruzione \textbf{esatta} (errore nullo).
\end{itemize}
\begin{alertblock}{Perché serve}
Ricalibrare l'intero modello a ogni chiusura trimestrale non è sostenibile: si
\textbf{congelano} i loadings stimati e si riutilizzano su curve successive.
\end{alertblock}
\end{frame}

\programmazioneslide{Proiettare una curva nuova sui loadings già stimati:
Passo 1 (scores $= V_k^T\mathbf{y}$) e Passo 2 (ricostruzione $=V_k\boldsymbol\alpha_k$) in R.}

\section{Conclusioni (\textasciitilde7 min)}
% ============================================================

\begin{frame}{Conclusioni}
\begin{itemize}
  \item La PCA sulle \textbf{variazioni mensili centrate} decompone la struttura a termine
        in componenti indipendenti; la frequenza mensile filtra il rumore di brevissimo
        periodo.
  \item Strumento matematico: la \textbf{SVD}; per Eckart--Young il troncamento a $k$
        componenti è l'approssimazione ottimale in norma di Frobenius.
  \item Le \textbf{prime tre componenti} spiegano il $99{,}51\%$ della varianza: da 20
        scadenze a 3 fattori, con meno dell'$1\%$ di informazione persa.
  \item I \textbf{loadings} (livello, pendenza, curvatura) e gli \textbf{scores} (quando si
        attivano) ricostruiscono la storia della curva alla luce degli eventi di politica
        monetaria.
  \item La \textbf{proiezione fuori campione} mostra che loadings calibrati a fine 2025
        restano validi, con soli 3 fattori, anche su curve di mesi successivi.
\end{itemize}
\end{frame}
